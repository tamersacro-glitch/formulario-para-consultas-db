-- 1. Crear base de datos
CREATE DATABASE gestion_escolar
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

-- 2. Seleccionar base de datos
USE gestion_escolar;


-- ==========================================
-- 3. TABLA ESTUDIANTE
-- ==========================================

CREATE TABLE estudiante (
    id_estudiante INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    apellido VARCHAR(100) NOT NULL,
    fecha_nacimiento DATE,
    correo VARCHAR(150) UNIQUE
);


-- ==========================================
-- 4. TABLA PROFESOR
-- ==========================================

CREATE TABLE profesor (
    id_profesor INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    apellido VARCHAR(100) NOT NULL,
    correo VARCHAR(150) UNIQUE,
    carrera VARCHAR(100)
);


-- ==========================================
-- 5. TABLA MATERIA
-- ==========================================

CREATE TABLE materia (
    id_materia INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    creditos INT NOT NULL,
    id_profesor INT,

    FOREIGN KEY (id_profesor)
        REFERENCES profesor(id_profesor)
);


-- ==========================================
-- 6. TABLA MATRÍCULA
-- ==========================================

CREATE TABLE matricula (
    id_matricula INT AUTO_INCREMENT PRIMARY KEY,
    id_estudiante INT NOT NULL,
    id_materia INT NOT NULL,
    periodo VARCHAR(50) NOT NULL,
    fecha_matricula DATE NOT NULL,
    estado VARCHAR(30) DEFAULT 'Activa',

    FOREIGN KEY (id_estudiante)
        REFERENCES estudiante(id_estudiante),

    FOREIGN KEY (id_materia)
        REFERENCES materia(id_materia)
);


-- ==========================================
-- 7. TABLA EVALUACIÓN
-- ==========================================

CREATE TABLE evaluacion (
    id_evaluacion INT AUTO_INCREMENT PRIMARY KEY,
    id_matricula INT NOT NULL,
    tipo VARCHAR(100) NOT NULL,
    calificacion DECIMAL(5,2) NOT NULL,
    fecha DATE NOT NULL,

    FOREIGN KEY (id_matricula)
        REFERENCES matricula(id_matricula)
);

-- ==========================================
-- 13. CONSULTA 1
-- ESTUDIANTES Y MATERIAS INSCRITAS
-- ==========================================
USE gestion_escolar;

SELECT
    estudiante.id_estudiante,
    estudiante.nombre,
    estudiante.apellido,
    materia.nombre AS materia,
    matricula.periodo,
    matricula.fecha_matricula
FROM matricula
INNER JOIN estudiante
    ON matricula.id_estudiante = estudiante.id_estudiante
INNER JOIN materia
    ON matricula.id_materia = materia.id_materia;


-- ==========================================
-- 14. CONSULTA 2
-- CALIFICACIONES DE LOS ESTUDIANTES
-- ==========================================
USE gestion_escolar;

SELECT
    estudiante.nombre,
    estudiante.apellido,
    materia.nombre AS materia,
    evaluacion.tipo,
    evaluacion.calificacion,
    evaluacion.fecha
FROM evaluacion
INNER JOIN matricula
    ON evaluacion.id_matricula = matricula.id_matricula
INNER JOIN estudiante
    ON matricula.id_estudiante = estudiante.id_estudiante
INNER JOIN materia
    ON matricula.id_materia = materia.id_materia
ORDER BY estudiante.apellido;

-- ==========================================
-- 17. CONSULTA ADICIONAL
-- MATERIAS Y PROFESORES
-- ==========================================
USE gestion_escolar;
SELECT
    materia.nombre AS materia,
    materia.creditos,
    profesor.nombre AS nombre_profesor,
    profesor.apellido AS apellido_profesor
FROM materia
INNER JOIN profesor
    ON materia.id_profesor = profesor.id_profesor;


-- ==========================================
-- 18. CONSULTA DE PROMEDIOS
-- ==========================================
USE gestion_escolar;
SELECT
    estudiante.nombre,
    estudiante.apellido,
    AVG(evaluacion.calificacion) AS promedio
FROM estudiante
INNER JOIN matricula
    ON estudiante.id_estudiante = matricula.id_estudiante
INNER JOIN evaluacion
    ON matricula.id_matricula = evaluacion.id_matricula
GROUP BY
    estudiante.id_estudiante,
    estudiante.nombre,
    estudiante.apellido;
